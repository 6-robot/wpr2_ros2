cmake_minimum_required(VERSION 3.8)
project(wpr2_bringup)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tf2_ros REQUIRED)

add_library(wpr2_serial STATIC src/driver/SerialCom.cpp src/driver/wpr2_driver.cpp)
target_include_directories(wpr2_serial PUBLIC include)
foreach(target wpr2_core wpr2_js_vel wpr2_lidar_filter wpr2_calibrate_velocity wpr2_test_motors wpr2_arms_test)
  add_executable(${target} src/${target}.cpp)
  ament_target_dependencies(${target} rclcpp geometry_msgs sensor_msgs std_msgs nav_msgs tf2_ros)
  install(TARGETS ${target} DESTINATION lib/${PROJECT_NAME})
endforeach()
target_link_libraries(wpr2_core wpr2_serial)
target_link_libraries(wpr2_test_motors wpr2_serial)
install(PROGRAMS scripts/vel_panel.py scripts/joints_panel.py scripts/wpr2_panel.py
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz scripts DESTINATION share/${PROJECT_NAME}
  PATTERN "__pycache__" EXCLUDE)
install(FILES README.md LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(interfaces test/test_interfaces.py TIMEOUT 120
    ENV "WPR2_BIN_DIR=${CMAKE_CURRENT_BINARY_DIR}" "ROS_LOCALHOST_ONLY=1")
endif()
ament_package()
