cmake_minimum_required(VERSION 3.8)
project(wpr2_behaviors)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp std_msgs geometry_msgs sensor_msgs visualization_msgs tf2_ros
    pcl_conversions pcl_ros wpr2_msgs)
  find_package(${dep} REQUIRED)
endforeach()
include_directories(include)
add_executable(wpr2_objects_3d src/objects_3d.cpp)
ament_target_dependencies(wpr2_objects_3d rclcpp std_msgs sensor_msgs geometry_msgs
  visualization_msgs tf2_ros pcl_conversions pcl_ros wpr2_msgs)
add_executable(wpr2_grab src/grab.cpp)
ament_target_dependencies(wpr2_grab rclcpp std_msgs sensor_msgs geometry_msgs wpr2_msgs)
install(TARGETS wpr2_objects_3d wpr2_grab DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME}
  PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
install(FILES README.md LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(behaviors test/test_behaviors.py TIMEOUT 180
    ENV BEHAVIORS_BIN_DIR=${CMAKE_CURRENT_BINARY_DIR} ROS_LOCALHOST_ONLY=1)
endif()
ament_package()
